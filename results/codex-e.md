import os, subprocess, datetime
from pathlib import Path
log=Path('/tmp/codex-e-bench.log').open('w', buffering=1)
for dataset in ['minio','kube-apiserver']:
 for threads in [1,8]:
  for rnd in range(1,6):
   stamp=subprocess.check_output(['date','-u','+%FT%TZ'],text=True).strip()
   print(f'ROUND dataset={dataset} threads={threads} round={rnd} utc={stamp} load={Path("/proc/loadavg").read_text().strip()}',file=log)
   variants=['old','aux','par']; variants=variants[(rnd-1)%3:]+variants[:(rnd-1)%3]
   for mode in variants:
    binary={'old':'p2lazy-old','aux':'p2lazy','par':'p2par'}[mode]
    cmd=['prlimit','--as=8589934592','--','build/'+binary,'/tmp/codex-e-data/'+dataset,'1']+(['--aux'] if mode=='aux' else [])
    print('COMMAND '+' '.join(cmd),file=log)
    result=subprocess.run(cmd,env=dict(os.environ,OMP_NUM_THREADS=str(threads)),text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    print(result.stdout,end='',file=log)
    if result.returncode or 'P2 OK' not in result.stdout: raise RuntimeError(result.stdout)
print(subprocess.check_output(['date','-u','+%FT%TZ'],text=True).strip()+' COMPLETE',file=log)
